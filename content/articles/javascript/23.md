### 模拟实现 for...of 遍历对象

```js
const p = {
    name: 'april',
    age: 18,
}

Object.defineProperty(p, Symbol.iterator, {
    enumerable: false,
    configurable: false,
    writable: false,
    value: function () {
        const _this = this
        let nowIndex = -1
        let key = Object.keys(_this)
        return {
            next: function () {
                nowIndex++
                return {
                    value: _this[key[nowIndex]],
                    done: nowIndex + 1 > key.length,
                }
            },
        }
    },
})
```

### 模拟实现 for 循环

```js
let students = {
    [Symbol.iterator]: function* () {
        for (let i = 0; i <= 100; i++) {
            yield i
        }
    },
}
for (const s of students) {
    console.log(s)
}
```
